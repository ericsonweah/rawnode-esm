```js
/** Simple comment mask (// and /* *\/) to avoid transforming commented-out code */
function computeCommentRanges(src) {
  const ranges = [];
  let i = 0, n = src.length, inSL = false, inML = false, str = null, tpl = false, esc = false;
  while (i < n) {
    const c = src[i], d = src[i + 1];
    if (str) { esc = !esc && c === "\\"; if (!esc && c === str) str = null; i++; continue; }
    if (tpl) {
      if (!esc && c === "`") { tpl = false; i++; continue; }
      esc = !esc && c === "\\"; i++; continue;
    }
    if (!inSL && !inML) {
      if (c === "'" || c === '"') { str = c; i++; continue; }
      if (c === "`") { tpl = true; i++; continue; }
      if (c === "/" && d === "/") { inSL = true; const s = i; i += 2; while (i < n && src[i] !== "\n") i++; ranges.push([s, i]); continue; }
      if (c === "/" && d === "*") { inML = true; const s = i; i += 2; while (i < n && !(src[i] === "*" && src[i+1] === "/")) i++; i = Math.min(n, i + 2); ranges.push([s, i]); continue; }
      i++; continue;
    } else if (inSL) {
      while (i < n && src[i] !== "\n") i++;
      inSL = false; continue;
    } else if (inML) {
      while (i < n && !(src[i] === "*" && src[i+1] === "/")) i++;
      i = Math.min(n, i + 2);
      inML = false; continue;
    }
  }
  return ranges;
}
function isInRanges(i, ranges) { for (const [s,e] of ranges) if (i >= s && i < e) return true; return false; }

/** Per-specifier namespace registry (stable, de-duplicated) */
function makeNSRegistry(topLevelImports) {
  const specToId = new Map();
  const posix = (s) => s.replace(/^node:/, "").replace(/^\.{1,2}\//, "").split("/").filter(Boolean).pop() || "mod";
  const slug = (s) => posix(s).replace(/[^A-Za-z0-9_]+/g, "_").replace(/^_+|_+$/g, "") || "mod";
  return {
    ensure(spec) {
      let id = specToId.get(spec);
      if (!id) {
        id = `__ns_${slug(spec)}`;
        specToId.set(spec, id);
        const line = `import * as ${id} from "${spec}";`;
        if (!topLevelImports.includes(line)) topLevelImports.push(line);
      }
      return id;
    },
    has(spec) { return specToId.has(spec); },
    id(spec) { return specToId.get(spec); },
  };
}
```

```js
if (true) {
```
