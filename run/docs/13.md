```js
// worker.mjs — RawNode ESM converter (worker side)
// ------------------------------------------------
// ✅ Handles alias destructuring like { promises: fs } = require("fs")
// ✅ Tracks converted, skipped, and errors
// ✅ Debug summaries and per-dir reports
// ✅ Zero dependencies, CSP-safe

import { parentPort } from "node:worker_threads";
import fs from "node:fs";
import path from "node:path";
import { performance } from "node:perf_hooks";

const DEBUG =
  process.argv.includes("--debug") || process.env.RAWNODE_ESM_DEBUG === "1";

const isTTY = process.stdout.isTTY;
function color(code) {
  return isTTY ? `\x1b[${code}m` : "";
}
const RESET = color(0);
const DIM = color("2");
const GREEN = color("32");
const CYAN = color("36");
const YELLOW = color("33");
const GRAY = color("90");

/*──────────────────────────────────────────────────────────────
 * MAIN WORKER HANDLER
 *──────────────────────────────────────────────────────────────*/
parentPort.on("message", async ({ dir, dryRun }) => {
  const start = performance.now();
  try {
    const { converted, skipped, errors, breakdown } = await replaceRequire(dir, dryRun);
    const durationMs = performance.now() - start;

    if (DEBUG) {
      const relDir = path.relative(process.cwd(), dir);
      console.log(
        `${CYAN}🧵 [Worker ${process.pid}]${RESET} ${relDir} — ${GREEN}${converted}${RESET} converted, ${GRAY}${skipped}${RESET} skipped, ${YELLOW}${errors}${RESET} errors in ${(durationMs / 1000).toFixed(2)} s`
      );
    }

    parentPort.postMessage({
      type: "done",
      dir,
      converted,
      skipped,
      errors,
      durationMs,
    });
  } catch (err) {
    const durationMs = performance.now() - start;
    parentPort.postMessage({
      type: "error",
      dir,
      durationMs,
      error: err.message,
      stack: err.stack,
    });
  }
});

/*──────────────────────────────────────────────────────────────
 * CORE CONVERSION LOGIC
 *──────────────────────────────────────────────────────────────*/
async function replaceRequire(dir, dryRun) {
  let convertedCount = 0;
  let skippedCount = 0;
  let errorCount = 0;
  const breakdown = { require: 0, exports: 0, moduleExports: 0 };

  function resolveImportPath(mod, baseDir) {
    const isRelative = mod.startsWith("./") || mod.startsWith("../");
    if (!isRelative) return mod; // core or package
    const abs = path.resolve(baseDir, mod);
    try {
      const stat = fs.statSync(abs);
      if (stat.isDirectory()) return `${mod.replace(/\/$/, "")}/index.js`;
    } catch {}
    return mod.endsWith(".js") ? mod : `${mod}.js`;
  }

  // detect function/class scopes for context awareness
  function detectScopes(code) {
    const ranges = [];
    const stack = [];
    for (let i = 0; i < code.length; i++) {
      const c = code[i];
      if (c === "{") stack.push(i);
      else if (c === "}") {
        const start = stack.pop();
        if (start !== undefined) ranges.push([start, i]);
      }
    }
    return ranges;
  }
  const isInside = (i, scopes) => scopes.some(([s, e]) => i > s && i < e);

  function walk(dirPath) {
    for (const file of fs.readdirSync(dirPath)) {
      const full = path.join(dirPath, file);
      const stat = fs.statSync(full);
      if (stat.isDirectory()) {
        walk(full);
        continue;
      }
      if (!file.endsWith(".js")) continue;

      try {
        let code = fs.readFileSync(full, "utf8");
        const original = code;
        const baseDir = path.dirname(full);
        const scopes = detectScopes(code);

        /*──────────────────────────────
         * REQUIRE → IMPORT REPLACEMENTS
         *──────────────────────────────*/

        // --- handle call requires: const x = require('x')(args)
        code = code.replace(
          /const\s+(\w+)\s*=\s*require\(['"]([^'"]+)['"]\)\(([^)]*)\);?/g,
          (_, name, mod, args) => {
            const importPath = resolveImportPath(mod, baseDir);
            return `import tmp_${name} from "${importPath}";\nconst ${name} = tmp_${name}(${args});`;
          }
        );

        // --- handle member access: const x = require('mod').member
        code = code.replace(
          /const\s+(\w+)\s*=\s*require\(['"]([^'"]+)['"]\)\.([A-Za-z$_][\w$]*)/g,
          (_, name, mod, member) => {
            const importPath = resolveImportPath(mod, baseDir);
            return `import * as __tmp_${name} from "${importPath}";\nconst ${name} = __tmp_${name}.${member};`;
          }
        );

        // --- handle alias destructuring like: const { promises: fs } = require("fs");
        code = code.replace(
          /const\s*\{\s*([A-Za-z$_][\w$]*)\s*:\s*([A-Za-z$_][\w$]*)\s*\}\s*=\s*require\(['"]([^'"]+)['"]\);?/g,
          (_, member, alias, mod) => {
            const importPath = resolveImportPath(mod, baseDir);
            return `import * as __tmp_${alias} from "${importPath}";\nconst ${alias} = __tmp_${alias}.${member};`;
          }
        );

        // --- regular destructuring: const { a,b } = require('x')
        code = code.replace(
          /const\s*\{\s*([^}]+)\s*\}\s*=\s*require\(['"]([^'"]+)['"]\);?/g,
          (_, names, mod) => {
            const importPath = resolveImportPath(mod, baseDir);
            return `import { ${names.trim()} } from "${importPath}";`;
          }
        );

        // --- simple require: const x = require('x')
        code = code.replace(
          /const\s+(\w+)\s*=\s*require\(['"]([^'"]+)['"]\);?/g,
          (_, name, mod) => {
            const importPath = resolveImportPath(mod, baseDir);
            return `import ${name} from "${importPath}";`;
          }
        );

        // --- module.exports = something
        code = code.replace(/module\.exports\s*=\s*([^;]+)/g, (_, rhs) => {
          breakdown.moduleExports++;
          return `export default ${rhs}`;
        });

        // --- exports.foo = ...
        code = code.replace(
          /\bexports\.([A-Za-z$_][\w$]*)\s*=\s*(?!require)([^;\n]+)/g,
          (_, key, rhs) => {
            breakdown.exports++;
            return `export const ${key} = ${rhs}`;
          }
        );

        // --- module.exports.foo = ...
        code = code.replace(
          /\bmodule\.exports\.([A-Za-z$_][\w$]*)\s*=\s*([^;\n]+)/g,
          (_, key, rhs) => {
            breakdown.exports++;
            return `export const ${key} = ${rhs}`;
          }
        );

        // --- dynamic require flag
        code = code.replace(
          /\brequire\(([^)]+)\)/g,
          (m) => `/* TODO dynamic require → await import(${m.slice(8, -1)}.js) */ ${m}`
        );

        /*──────────────────────────────
         * RESULTS
         *──────────────────────────────*/
        if (code !== original) {
          convertedCount++;
          if (!dryRun) fs.writeFileSync(full, code, "utf8");
          if (DEBUG) {
            const rel = path.relative(process.cwd(), full);
            console.log(`${DIM}🧩 Converted:${RESET} ${rel}`);
          }
        } else {
          skippedCount++;
        }
      } catch (err) {
        errorCount++;
        if (DEBUG) {
          const rel = path.relative(process.cwd(), file);
          console.log(`${YELLOW}⚠️  Error in ${rel}:${RESET} ${err.message}`);
        }
      }
    }
  }

  walk(dir);
  return { converted: convertedCount, skipped: skippedCount, errors: errorCount, breakdown };
}
```
