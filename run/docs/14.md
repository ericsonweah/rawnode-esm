```js
  // UPDATED: smarter .js / index.js resolver with fallback (Case 1, Case 2, default→Case 2)
  function resolveImportPath(mod, baseDir) {
    const isRelative = mod.startsWith("./") || mod.startsWith("../");
    if (!isRelative) return mod; // Node builtins or external packages

    const abs = path.resolve(baseDir, mod);
    const filePath = `${abs}.js`;
    const indexPath = path.join(abs, "index.js");

    try {
      // Case 1 — direct file (./foo.js)
      if (fs.existsSync(filePath) && fs.statSync(filePath).isFile()) {
        return `${mod}.js`;
      }
      // Case 2 — directory with index.js (./foo/index.js)
      if (fs.existsSync(indexPath) && fs.statSync(indexPath).isFile()) {
        return `${mod.replace(/\/$/, "")}/index.js`;
      }
      // Default fallback → assume Case 2 when nothing found
      return `${mod.replace(/\/$/, "")}/index.js`;
    } catch {
      // Graceful fallback on any fs error → Case 2
      return `${mod.replace(/\/$/, "")}/index.js`;
    }
  }
```

### 🧭 **Design Behavior Summary**

| Situation                                      | Result                           | Comment                |
| ---------------------------------------------- | -------------------------------- | ---------------------- |
| `./foo.js` exists                              | `import … from "./foo.js"`       | **Case 1**             |
| `./foo/index.js` exists                        | `import … from "./foo/index.js"` | **Case 2**             |
| Neither found                                  | `import … from "./foo/index.js"` | **Default fallback**   |
| Path is non-relative (`fs`, `http`, `express`) | returned unchanged               | builtin / package skip |

---

### 🧠 **Why This Works**

* Uses `fs.existsSync` + `fs.statSync` (synchronous, low-overhead in worker threads).
* Resolves ambiguity safely without blocking the event loop in the main process.
* Always returns an explicit `.js` or `/index.js` extension to stay ESM-correct.
* Defaults deterministically to Case 2 so conversion never breaks mid-run.
