  async function resolveImportPath(mod, baseDir) {
    const isRelative = mod.startsWith("./") || mod.startsWith("../");
    if (!isRelative) return mod; // Node builtins or external packages

    const abs = path.resolve(baseDir, mod);
    const filePath = `${abs}.js`;
    const indexPath = path.join(abs, "index.js");

    try {
      // Case 1 — check for direct file
      await fs.promises.access(filePath, fs.constants.F_OK);
      const stat = await fs.promises.stat(filePath);
      if (stat.isFile()) return `${mod}.js`;
    } catch { /* not found */ }

    try {
      // Case 2 — check for index.js inside directory
      await fs.promises.access(indexPath, fs.constants.F_OK);
      const stat = await fs.promises.stat(indexPath);
      if (stat.isFile()) return `${mod.replace(/\/$/, "")}/index.js`;
    } catch { /* not found */ }

    // Default fallback → assume Case 2
    return `${mod.replace(/\/$/, "")}/index.js`;
  }
