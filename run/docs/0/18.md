// UPDATED: adaptive non-blocking walker with live debug diagnostics
function walk(dirPath) {
  const DEBUG = process.argv.includes("--debug-walk"); // enable with flag
  const queue = [dirPath];
  let processed = 0;
  let batchSize = 10;               // initial batch size
  let avgMsPerBatch = 0;
  const targetMs = 8;               // target budget per slice (~1 frame)
  const minBatch = 5;
  const maxBatch = 100;
  let lastLog = 0;

  // ANSI helpers (color output only if TTY)
  const isTTY = process.stdout.isTTY;
  const color = (c) => (isTTY ? `\x1b[${c}m` : "");
  const RESET = color(0);
  const CYAN = color("36");
  const GREEN = color("32");
  const YELLOW = color("33");
  const MAGENTA = color("35");
  const DIM = color("2");

  function processNext() {
    const start = performance.now();
    let batch = 0;

    while (queue.length && batch++ < batchSize) {
      const currentDir = queue.pop();
      let files;
      try {
        files = fs.readdirSync(currentDir);
      } catch {
        continue;
      }

      for (const file of files) {
        const full = path.join(currentDir, file);
        let stat;
        try {
          stat = fs.statSync(full);
        } catch {
          continue;
        }

        if (stat.isDirectory()) {
          queue.push(full);
          continue;
        }
        if (!file.endsWith(".js")) continue;

        processed++;
        // ─────────────────────────────────────────────
        // existing synchronous conversion logic
        // ─────────────────────────────────────────────
        let code = fs.readFileSync(full, "utf8");
        const original = code;
        const baseDir = path.dirname(full);
        const scopes = detectScopes(code);
        const topLevelImports = [];
        const seenImports = new Set();

        // require('mod')(args)
        code = code.replace(
          /const\s+(\w+)\s*=\s*require\(['"]([^'"]+)['"]\)\(([^)]*)\);?/g,
          (m, name, mod, args, idx) => {
            const imp = resolveImportPath(mod, baseDir);
            if (isInside(idx, scopes))
              return `const ${name} = await (async()=> (await import("${imp}")).default(${args}))();`;
            topLevelImports.push(`import tmp_${name} from "${imp}";`);
            return `const ${name} = tmp_${name}(${args});`;
          }
        );

        // require('mod').member or call
        code = code.replace(
          /const\s+(\w+)\s*=\s*require\(['"]([^'"]+)['"]\)\.([A-Za-z$_][\w$]*)(\([^)]*\))?/g,
          (m, name, mod, member, call = "", idx) => {
            const imp = resolveImportPath(mod, baseDir);
            if (isInside(idx, scopes))
              return `const ${name} = await (async()=> (await import("${imp}")).${member}${call})();`;
            topLevelImports.push(`import * as __tmp_${name} from "${imp}";`);
            return `const ${name} = __tmp_${name}.${member}${call};`;
          }
        );

        // const x = require('mod')
        code = code.replace(
          /const\s+(\w+)\s*=\s*require\(['"]([^'"]+)['"]\);?/g,
          (m, name, mod, idx) => {
            const imp = resolveImportPath(mod, baseDir);
            if (isInside(idx, scopes))
              return `const ${name} = await (async()=> await import("${imp}"))();`;
            topLevelImports.push(`import ${name} from "${imp}";`);
            return `// moved import for ${name}`;
          }
        );

        // alias destructuring: const { promises: fs } = require("fs");
        code = code.replace(
          /const\s*\{\s*([A-Za-z$_][\w$]*)\s*:\s*([A-Za-z$_][\w$]*)\s*\}\s*=\s*require\(['"]([^'"]+)['"]\);?/g,
          (m, member, alias, mod, idx) => {
            const imp = resolveImportPath(mod, baseDir);
            if (isInside(idx, scopes))
              return `const ${alias} = await (async()=> (await import("${imp}")).${member})()`;
            topLevelImports.push(`import * as __tmp_${alias} from "${imp}";`);
            return `// moved import for ${alias}\nconst ${alias} = __tmp_${alias}.${member};`;
          }
        );

        // destructured require
        code = code.replace(
          /const\s*\{\s*([^}]+)\s*\}\s*=\s*require\(['"]([^'"]+)['"]\);?/g,
          (m, names, mod, idx) => {
            const imp = resolveImportPath(mod, baseDir);
            if (isInside(idx, scopes))
              return `const { ${names.trim()} } = await (async()=> await import("${imp}"))();`;
            topLevelImports.push(`import { ${names.trim()} } from "${imp}";`);
            return `// moved import for { ${names.trim()} }`;
          }
        );

        let hasDefault = false;
        code = code.replace(/module\.exports\s*=\s*([^;]+)/g, (_, rhs) => {
          hasDefault = true;
          return `export default ${rhs}`;
        });
        code = code.replace(
          /\bexports\.([A-Za-z$_][\w$]*)\s*=\s*(?!require)([^;\n]+)/g,
          (_, key, rhs) => `export const ${key} = ${rhs}`
        );
        code = code.replace(
          /\bmodule\.exports\.([A-Za-z$_][\w$]*)\s*=\s*([^;\n]+)/g,
          (_, key, rhs) =>
            hasDefault ? `export { ${rhs} as ${key} };` : `export const ${key} = ${rhs}`
        );

        code = code.replace(
          /\brequire\(([^)"']+)\)/g,
          (m) => `/* TODO dynamic require → await import(${m.slice(8, -1)}.js) */ ${m}`
        );

        if (topLevelImports.length) {
          const unique = topLevelImports.filter((imp) => {
            if (seenImports.has(imp)) return false;
            seenImports.add(imp);
            return true;
          });
          code = `${unique.join("\n")}\n\n${code}`;
        }

        if (code !== original) {
          count++;
          if (!dryRun) fs.writeFileSync(full, code, "utf8");
        }
      }
    }

    // Adaptive pacing
    const elapsed = performance.now() - start;
    avgMsPerBatch = avgMsPerBatch * 0.8 + elapsed * 0.2;
    if (avgMsPerBatch > targetMs && batchSize > minBatch) {
      batchSize = Math.max(minBatch, Math.floor(batchSize * 0.9));
    } else if (avgMsPerBatch < targetMs * 0.6 && batchSize < maxBatch) {
      batchSize = Math.min(maxBatch, Math.floor(batchSize * 1.1));
    }

    // Debug output every ~500ms
    if (DEBUG && performance.now() - lastLog > 500) {
      const pct = Math.min(100, (batchSize / maxBatch) * 100);
      const colorize =
        pct > 80 ? GREEN : pct > 40 ? YELLOW : MAGENTA;
      const throughput = processed / ((performance.now() - start) / 1000 + 0.001);
      console.log(
        `${DIM}[worker:${process.pid}]${RESET} ${CYAN}batch${RESET}=${colorize}${batchSize}${RESET}  ` +
        `${DIM}avg=${avgMsPerBatch.toFixed(2)}ms${RESET}  ` +
        `${DIM}q=${queue.length}${RESET}  ` +
        `${DIM}→ ${throughput.toFixed(1)} files/s${RESET}`
      );
      lastLog = performance.now();
    }

    if (queue.length) queueMicrotask(processNext);
  }

  processNext();
}
