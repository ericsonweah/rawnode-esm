## 🧠 `/src/modules/rawnode-esm/run.mjs`

```js
#!/usr/bin/env node
// run.mjs — main CLI & worker pool controller (fixed version)

import { Worker } from "node:worker_threads";
import { cpus } from "node:os";
import fs from "node:fs";
import path from "node:path";
import { performance } from "node:perf_hooks";

/* ────────────────────────────────────────────────────────────────
 * CLI flags
 * ──────────────────────────────────────────────────────────────── */
const args = process.argv.slice(2);
const dryRun = args.includes("--dry-run");
const targetIndex = args.indexOf("--target");
const concurrencyIndex = args.indexOf("--concurrency");

const targetDir =
  targetIndex !== -1 && args[targetIndex + 1]
    ? path.resolve(args[targetIndex + 1])
    : path.resolve("./src");

const concurrency =
  concurrencyIndex !== -1 && args[concurrencyIndex + 1]
    ? Math.max(1, Number(args[concurrencyIndex + 1]))
    : Math.max(1, Math.floor(cpus().length / 2));

console.log(
  `🚀 Starting RawNode ESM conversion\n` +
    `Target: ${targetDir}\n` +
    `Concurrency: ${concurrency}\n` +
    (dryRun ? "Mode: 🔍 Dry-run (no writes)\n" : "Mode: ✍️  Write in-place\n")
);

/* ────────────────────────────────────────────────────────────────
 * Discover work items (subdirs + .js files)
 * ──────────────────────────────────────────────────────────────── */
function discoverTargets(root) {
  const items = [];
  for (const entry of fs.readdirSync(root, { withFileTypes: true })) {
    const full = path.join(root, entry.name);
    if (entry.isDirectory()) {
      items.push(full);
    } else if (entry.isFile() && entry.name.endsWith(".js")) {
      items.push(full);
    }
  }
  return items;
}

const dirs = discoverTargets(targetDir);
let nextJob = 0;
let completed = 0;
let totalFiles = 0;
const start = performance.now();

/* ────────────────────────────────────────────────────────────────
 * Worker management
 * ──────────────────────────────────────────────────────────────── */
const workers = [];
const results = [];

function createWorker(id) {
  const worker = new Worker(new URL("./worker.mjs", import.meta.url));
  worker.idle = false;

  worker.on("message", (msg) => {
    switch (msg.type) {
      case "log":
        process.stdout.write(`🧵 Worker ${id} → ${msg.message}\n`);
        break;
      case "done":
        completed++;
        totalFiles += msg.converted;
        process.stdout.write(
          `✅ Worker ${id} → done ${path.basename(msg.dir)} (${msg.converted} files)\n`
        );
        results.push(msg);
        assignNext(worker, id);
        break;
      case "error":
        process.stdout.write(`⚠️  Worker ${id} → error in ${msg.dir}: ${msg.error}\n`);
        assignNext(worker, id);
        break;
    }
  });

  worker.on("exit", (code) => {
    if (code !== 0)
      console.error(`❌ Worker ${id} exited with code ${code}`);
    // Check if all workers are done after exit
    if (workers.every((w) => w.idle || w.threadId === worker.threadId)) {
      finish();
    }
  });

  return worker;
}

function assignNext(worker, id) {
  if (nextJob >= dirs.length) {
    worker.idle = true;
    if (workers.every((w) => w.idle)) {
      finish();
    }
    return;
  }

  const dir = dirs[nextJob++];
  worker.idle = false;
  worker.postMessage({ dir, dryRun });
  process.stdout.write(`🧵 Worker ${id} → processing ${path.basename(dir)}\n`);
}

function finish() {
  const end = performance.now();
  const elapsed = ((end - start) / 1000).toFixed(2);
  console.log(
    `\n🎉 All workers finished\n` +
      `Modules/files processed: ${completed}\n` +
      `Total files converted: ${totalFiles}\n` +
      `Elapsed: ${elapsed}s\n`
  );
  process.exit(0);
}

/* ────────────────────────────────────────────────────────────────
 * Start pool
 * ──────────────────────────────────────────────────────────────── */
if (dirs.length === 0) {
  console.log("⚠️  No .js files or directories found in target.");
  process.exit(0);
}

for (let i = 0; i < Math.min(concurrency, dirs.length); i++) {
  const worker = createWorker(i + 1);
  workers.push(worker);
}

workers.forEach((w, i) => assignNext(w, i + 1));
```

### ✅ What’s fixed

| Problem                                      | Solution                                            |
| -------------------------------------------- | --------------------------------------------------- |
| Converter skipped `.js` files in target root | Now includes both `.js` files and subdirectories    |
| Process never exited                         | `finish()` now called once all workers idle or exit |
| Wrong target name in logs                    | Now uses `path.basename()` for clarity              |
| Accurate total + elapsed time                | Added runtime timer summary                         |
