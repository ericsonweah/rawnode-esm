## ✅ `/src/modules/rawnode-esm/run.mjs`

```js
#!/usr/bin/env node
/**
 * RawNode-ESM CLI — cinematic, zero-dependency CommonJS→ESM converter
 * ---------------------------------------------------------------
 * Features:
 *  - Parallel worker pool (adaptive concurrency)
 *  - Gradient progress bar (🟩→🟨→🟥)
 *  - ETA, throughput, and live updates
 *  - Quiet & JSON report modes
 *  - Tracks converted, skipped, and error counts
 *  - CSP-safe, no dependencies
 */

import fs from "node:fs";
import path from "node:path";
import os from "node:os";
import { Worker } from "node:worker_threads";
import { performance } from "node:perf_hooks";

/*──────────────────────────────────────────────────────────────
 * CLI ARGUMENTS
 *──────────────────────────────────────────────────────────────*/
const args = process.argv.slice(2);
const getArg = (flag, def = null) => {
  const idx = args.indexOf(flag);
  return idx !== -1 ? args[idx + 1] : def;
};

const target = getArg("--target", ".");
const concurrency = parseInt(getArg("--concurrency", os.cpus().length)) || 6;
const dryRun = args.includes("--dry-run");
const quiet = args.includes("--quiet");
const jsonReport = args.includes("--json-report");
const reportPathArg = getArg("--report", "./rawnode-esm-report.json");
const timestamped = reportPathArg.includes("%DATE%");
const debug = args.includes("--debug") || process.env.RAWNODE_ESM_DEBUG === "1";

/*──────────────────────────────────────────────────────────────
 * UTILITIES
 *──────────────────────────────────────────────────────────────*/
const isTTY = process.stdout.isTTY;
const color = (c) => (isTTY ? `\x1b[${c}m` : "");
const RESET = color(0);
const GREEN = color("32");
const YELLOW = color("33");
const RED = color("31");
const CYAN = color("36");
const DIM = color("2");

function gradientColor(pct) {
  if (!isTTY) return "";
  if (pct < 0.5) return `\x1b[32m`; // green
  if (pct < 0.8) return `\x1b[33m`; // yellow
  return `\x1b[31m`; // red
}

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

/*──────────────────────────────────────────────────────────────
 * DISCOVER DIRECTORIES
 *──────────────────────────────────────────────────────────────*/
function findDirs(base) {
  const dirs = [];
  (function walk(p) {
    const entries = fs.readdirSync(p, { withFileTypes: true });
    let hasJs = false;
    for (const e of entries) {
      const full = path.join(p, e.name);
      if (e.isDirectory()) walk(full);
      else if (e.isFile() && e.name.endsWith(".js")) hasJs = true;
    }
    if (hasJs) dirs.push(p);
  })(path.resolve(base));
  return dirs;
}

/*──────────────────────────────────────────────────────────────
 * MAIN DRIVER
 *──────────────────────────────────────────────────────────────*/
(async function main() {
  console.log(`${CYAN}🚀 Starting RawNode ESM conversion${RESET}`);
  console.log(`Target: ${target}`);
  console.log(`Concurrency: ${concurrency}`);
  console.log(`Mode: ${dryRun ? "🔍 Dry-run preview" : "✍️  Write in-place"}`);
  if (quiet) console.log(`${DIM}(Quiet mode enabled)${RESET}\n`);

  const dirs = findDirs(target);
  const totalDirs = dirs.length;
  const totalFilesEstimate = dirs.length * 10; // heuristic for ETA

  const pool = [];
  const results = [];
  let active = 0;
  let completed = 0;

  let totalConverted = 0;
  let totalSkipped = 0;
  let totalErrors = 0;

  const start = performance.now();

  function renderProgress() {
    if (quiet || !isTTY) return;
    const pct = completed / totalDirs;
    const barWidth = Math.max(10, Math.min(process.stdout.columns - 40, 40));
    const filled = Math.round(pct * barWidth);
    const col = gradientColor(pct);
    const bar =
      col +
      "█".repeat(filled) +
      color("90") +
      "░".repeat(barWidth - filled) +
      RESET;
    const percent = (pct * 100).toFixed(1).padStart(5);
    const elapsed = (performance.now() - start) / 1000;
    const rate = ((completed || 1) / elapsed).toFixed(1);
    const eta = ((totalDirs - completed) / (completed / elapsed || 1)).toFixed(1);

    process.stdout.write(
      `\r${bar} ${percent}% | ${completed}/${totalDirs} dirs | ${rate} dirs/s | ETA ${eta}s `
    );
  }

  async function runWorker(id, dir) {
    return new Promise((resolve) => {
      const worker = new Worker(new URL("./worker.mjs", import.meta.url), {
        workerData: null,
        argv: debug ? ["--debug"] : [],
      });
      const startTime = performance.now();

      worker.on("message", (msg) => {
        if (msg.type === "done") {
          const { converted, skipped, errors, durationMs } = msg;
          totalConverted += converted;
          totalSkipped += skipped;
          totalErrors += errors;
          results.push({ id, dir, converted, skipped, errors, durationMs });
        } else if (msg.type === "error") {
          totalErrors++;
          results.push({ id, dir, error: msg.error, durationMs: msg.durationMs });
        }
      });

      worker.on("exit", () => {
        completed++;
        renderProgress();
        resolve();
      });

      worker.postMessage({ dir, dryRun });
    });
  }

  // Launch workers in batches
  async function processAll() {
    for (const dir of dirs) {
      while (active >= concurrency) await sleep(50);
      active++;
      runWorker(active, dir).then(() => (active--));
      renderProgress();
    }

    while (active > 0) {
      renderProgress();
      await sleep(100);
    }
  }

  // Progress loop
  const progressInterval = isTTY ? setInterval(renderProgress, 200) : null;
  await processAll();
  if (progressInterval) clearInterval(progressInterval);

  const elapsed = (performance.now() - start) / 1000;

  // Final full-green bar
  if (!quiet && isTTY) {
    const barWidth = Math.max(10, Math.min(process.stdout.columns - 40, 40));
    const bar = GREEN + "█".repeat(barWidth) + RESET;
    process.stdout.write(
      `\r${bar} 100.0% | ${totalDirs}/${totalDirs} dirs | ${(
        totalDirs / elapsed
      ).toFixed(1)} dirs/s | Done in ${elapsed.toFixed(2)}s\n`
    );
  }

  console.log(
    `\n${GREEN}✅ ${totalConverted}${RESET} converted, ${YELLOW}${totalSkipped}${RESET} skipped, ${RED}${totalErrors}${RESET} errors in ${elapsed.toFixed(
      2
    )}s`
  );

  /*──────────────────────────────────────────────────────────────
   * JSON REPORT
   *──────────────────────────────────────────────────────────────*/
  if (jsonReport) {
    const reportPath = timestamped
      ? reportPathArg.replace(
          "%DATE%",
          new Date().toISOString().replace(/[:.]/g, "-")
        )
      : reportPathArg;

    const data = {
      timestamp: new Date().toISOString(),
      config: { target, concurrency, dryRun, quiet },
      totals: { converted: totalConverted, skipped: totalSkipped, errors: totalErrors, elapsed },
      workers: results,
    };
    fs.mkdirSync(path.dirname(reportPath), { recursive: true });
    fs.writeFileSync(reportPath, JSON.stringify(data, null, 2));
    console.log(`${CYAN}🧾 Report written to:${RESET} ${reportPath}`);
  }

  console.log(`${CYAN}✨ All done.${RESET}`);
})();
```
